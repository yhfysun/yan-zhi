替换只改 #s_lg_img 的图源，版式与搜索框位置不变；建议同时关掉 hover 换脸，否则鼠标移上会变回百度。

可粘贴到百度首页 DevTools Console 的替换脚本（刷新即还原）

(function () {
  var img = document.getElementById('s_lg_img');
  if (!img) { console.warn('#s_lg_img 未找到，页面结构可能已变'); return; }

  // 天猫 logo（内联 SVG，自包含、不依赖外链、可离线）
  var svg = '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 270 129">'
    + '<g>'
    + '<g transform="translate(6,10)">'
    + '<path d="M6 30 L6 6 L30 20 H70 L94 6 V30 c10 8 16 20 16 33 0 24-25 43-55 43S6 87 6 63c0-13 6-25 16-33z" fill="#111"/>'
    + '<ellipse cx="30" cy="58" rx="8" ry="9.5" fill="#fff"/><ellipse cx="66" cy="58" rx="8" ry="9.5" fill="#fff"/>'
    + '<ellipse cx="31" cy="60" rx="3.4" ry="4.6" fill="#111"/><ellipse cx="67" cy="60" rx="3.4" ry="4.6" fill="#111"/>'
    + '<path d="M44 74 l7 5 -7 5 -7 -5z" fill="#FF0036"/>'
    + '</g>'
    + '<text x="112" y="80" font-family="PingFang SC,Microsoft YaHei,sans-serif" font-weight="800" font-size="62" fill="#FF0036">天猫</text>'
    + '<text x="116" y="104" font-family="Arial,sans-serif" font-weight="700" font-size="15" fill="#111" letter-spacing="5">TMALL</text>'
    + '</g></svg>';

  img.src = 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svg);
  img.style.width = '270px';
  img.style.height = '129px';

  // 关掉 hover 换脸（百度 logo 悬停会切另一张）
  var lg = document.getElementById('lg') || img.parentElement;
  if (lg) { lg.style.backgroundImage = 'none'; }
  img.onmouseenter = null; img.onmouseover = null;

  console.log('✅ #s_lg_img 已换成天猫 logo（刷新页面即还原）');
})();
想换成官方天猫图，只要把上面 img.src 换成你手上的天猫 logo 图片地址即可（要求 270×129 比例，透明底更好）。
