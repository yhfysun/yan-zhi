export async function run(task) {
  const plan = await planOf(task);
  for (const step of plan.steps) {
    // 执行并回传进度
    await step.run();
    emit('progress', step.id);
  }
  return plan.result;
}

这次改的关键点
旧问题：双指一碰就先发了 left down，之后只补 up 的条件不成立，远端左键一直按着 → 之后所有单击都变成拖拽，体感「单击没用」。
新做法：点按直接发原子 click，不再赌 down+up；进双指/取消/切后台/断开都强制补抬左键。
手势优先级：单指位移 > 8px 才算拖动；双指先锁意图（间距变化→缩放，中点位移→滚动），不会一边滚一边放大。
双模式并行（UU 同款）：画面直接点按始终可用（点哪点哪），触控板同时可拖动指针/点按左键/双指滚动，不需要切换。
触控板：指哪打哪的自绘光标 + 大鼠标图案 + 左右键 + 滚动条；拖动时板面高亮、触点跟随，远端 hover 实时生效。
可点区域：整块画面滚轮事件也可直接试（触控板双指）
