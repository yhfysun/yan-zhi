# 可直接导出/导入，也是喂给大模型的形态
version: 1
datasource: ds_project
code: order
name: 订单
domain: 交易域
description: >
  一笔订单 = 一个用户一次下单行为。金额口径统一为「实付金额」。

entities:
  - name: order   type: primary  expr: order_id
  - name: user    type: foreign  expr: user_id

time_dimensions:
  - name: created_at
    description: 下单时间
    synonyms: [下单日期, 创建时间]
    granularities: [day, week, month, quarter, year]

dimensions:
  - name: region
    description: 用户所属大区
    ref_attr: region
    sample_values: [华东, 华北, 华南]
  - name: status
    description: 订单当前状态
    ref_attr: order_status      # 枚举 + 状态机来自标准属性

measures:
  - name: order_count      agg: count_distinct  additive: additive
  - name: pay_amount_sum   agg: sum             additive: additive
  - name: avg_order_value  expr: pay_amount_sum/order_count
    additive: non_additive     # 禁止上卷求和

relations:
  - left: order  right: user  cardinality: many_to_one
    on: order.user_id = user.user_id  status: confirmed

SELECT
  __u.name                              AS user_name,
  COUNT(DISTINCT __o.order_id)      AS order_count,
  SUM(__o.pay_amount)                  AS pay_amount_sum
FROM (
  SELECT o.id AS order_id, o.user_id AS user_id,
         u.name AS user_name, u.region AS region,
         o.pay_amount AS pay_amount, o.status AS status,
         o.created_at AS created_at
  FROM t_order o
  LEFT JOIN t_user u ON u.id = o.user_id
  WHERE o.is_deleted = 0
) __o
LEFT JOIN ( /* user */ ) __u ON __u.user_id = __o.user_id
WHERE __u.region = '华东'
  AND __o.created_at >= date('now', '-1 month')
GROUP BY __u.name
ORDER BY pay_amount_sum DESC
LIMIT 50
